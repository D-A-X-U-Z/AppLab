Guía Rápida de Despliegue en XAMPP (C:\xampp\htdocs\utvm_reservas_labs)

  1. Abra el Panel de Control de XAMPP e inicie los módulos Apache y MySQL (Puerto 3306).
  2. Copie esta carpeta del proyecto dentro de C:\xampp\htdocs\utvm_reservas_labs.
  3. Abra http://localhost/phpmyadmin e importe el archivo database/utvm_reservas_labs.sql (o simplemente abra http://localhost/utvm_reservas_labs/ y el backend api/conexion.php auto-aprovisionará la base de datos automáticamente).

Estructura SQL de las 4 Tablas (database/utvm_reservas_labs.sql):

CREATE DATABASE IF NOT EXISTS `utvm_reservas_labs` DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE `utvm_reservas_labs`;

-- 1. Tabla: usuarios_admin
CREATE TABLE `usuarios_admin` (
  `id_usuario` INT AUTO_INCREMENT PRIMARY KEY,
  `nombre_completo` VARCHAR(120) NOT NULL,
  `correo` VARCHAR(120) NOT NULL UNIQUE,
  `password_hash` VARCHAR(255) NOT NULL,
  `rol` ENUM('Administrador', 'Encargado') NOT NULL DEFAULT 'Encargado',
  `activo` TINYINT(1) NOT NULL DEFAULT 1
);

-- 2. Tabla: laboratorios
CREATE TABLE `laboratorios` (
  `id_laboratorio` INT AUTO_INCREMENT PRIMARY KEY,
  `nombre` VARCHAR(120) NOT NULL,
  `edificio` VARCHAR(60) NOT NULL,
  `planta` VARCHAR(60) NOT NULL,
  `capacidad_maxima` INT NOT NULL DEFAULT 25,
  `cupo_ocupado_actual` INT NOT NULL DEFAULT 0,
  `horario_atencion` VARCHAR(80) NOT NULL,
  `inventario_equipos` TEXT NOT NULL,
  `requisitos_obligatorios` TEXT NOT NULL,
  `estatus_operativo` ENUM('Disponible', 'Saturado', 'En Mantenimiento') NOT NULL DEFAULT 'Disponible',
  `id_encargado` INT DEFAULT NULL,
  FOREIGN KEY (`id_encargado`) REFERENCES `usuarios_admin`(`id_usuario`)
);

-- 3. Tabla: reservaciones
CREATE TABLE `reservaciones` (
  `id_reservacion` INT AUTO_INCREMENT PRIMARY KEY,
  `folio` VARCHAR(30) DEFAULT NULL UNIQUE,
  `matricula_estudiante` VARCHAR(20) NOT NULL,
  `nombre_estudiante` VARCHAR(120) NOT NULL,
  `carrera` VARCHAR(120) NOT NULL,
  `id_laboratorio` INT NOT NULL,
  `fecha_reserva` DATE NOT NULL,
  `bloque_horario` VARCHAR(50) NOT NULL,
  `motivo_uso` TEXT NOT NULL,
  `acepto_requisitos` TINYINT(1) NOT NULL DEFAULT 1,
  `estado` ENUM('Pendiente', 'Confirmado', 'Asistio', 'Cancelado') NOT NULL DEFAULT 'Pendiente',
  `motivo_cancelacion` TEXT DEFAULT NULL,
  `codigo_qr_hash` VARCHAR(150) DEFAULT NULL,
  `fecha_solicitud` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (`id_laboratorio`) REFERENCES `laboratorios`(`id_laboratorio`)
);

-- 4. Tabla: bitacora_accesos
CREATE TABLE `bitacora_accesos` (
  `id_acceso` INT AUTO_INCREMENT PRIMARY KEY,
  `id_reservacion` INT DEFAULT NULL,
  `folio_validado` VARCHAR(30) NOT NULL,
  `id_encargado_valido` INT DEFAULT NULL,
  `metodo_validacion` ENUM('Escaner_QR', 'Folio_Manual') NOT NULL DEFAULT 'Escaner_QR',
  `resultado_validacion` ENUM('Valido', 'Invalido', 'Ya_Utilizado') NOT NULL DEFAULT 'Valido',
  `fecha_hora_ingreso` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (`id_reservacion`) REFERENCES `reservaciones`(`id_reservacion`),
  FOREIGN KEY (`id_encargado_valido`) REFERENCES `usuarios_admin`(`id_usuario`)
);
